property get NextCourtesyContact(): Date{
    var nextCourtesyDate: Date = null

    if(this.LastCourtesyContact_Ext != null) {
       nextCourtesyDate = this.LastCourtesyContact_Ext.addMonths(6)
    }

    return nextCourtesyDate
}


What this code does ?

This getter property calculates a next courtesy contact date based on an existing date.
If LastCourtesyContact_Ext has a value, it returns a new date six months later.
If LastCourtesyContact_Ext is null, it safely returns null.
The value is computed dynamically and is not stored in the database.

uses entity.Activity
uses gw.api.util.DateUtil
uses gw.transaction.Transaction

function createSimpleActivity(claim: entity.Claim) : Activity {

  var bundle = Transaction.getCurrent()

  var act = new Activity()
  act.Claim = claim
  act.Subject = "Call customer to confirm address"
  act.Description = "Customer's address looks incomplete. Please call and confirm."
  act.Priority = Priority.TC_NORMAL
  act.TargetDate = DateUtil.currentDate().addDays(2)
  act.AssignedGroup = claim.AssignedGroup
  act.AssignedUser = claim.AssignedUser
  
  claim.addToActivities(act)
  
  return act //Transaction.commit()
}

funtion readValue(claims: List<String>, index: int) : String {
	
	try{
		print("Trying to read values...")
		return claims[index]
	}
	
	catch(e: IndexOuOfBoundException){
		print("Invalid index: " + index)
		return "NOT_FOUND"
	}
	
	catch(e: Exception){
		print("Some other unexcepted error occurred: " e.Message)
		return "ERROR"
	}

	finally{
		print("process completed")
	}

}

uses gw.transaction.Transaction
uses gw.api.database.Query

function createPropertyIncident(claimNumber : String) : entity.PropertyIncident {
  var out : entity.PropertyIncident = null

  Transaction.runWithNewBundle(\ b -> {
    var claim = Query.make(entity.Claim)
      .compare("ClaimNumber", Equals, claimNumber)
      .select().atMostOneRow()

    if (claim == null) {
      throw new IllegalArgumentException("No claim found for " + claimNumber)
    }

    var pin = new entity.PropertyIncident(b)
    pin.Claim = claim
    pin.Description = "Property incident (Fire) created via script"
    pin.IncidentDate = gw.api.util.DateUtil.currentDate()
    pin.PropertyDescription = "Single-family dwelling"
    pin.LossCause = typekey.LossCause.FIRE  // adjust if your typelist differs

    var addr = new entity.Address(b)
    addr.AddressLine1 = "221B Baker Street"
    addr.City = "Kolkata"
    addr.State = typekey.State.WB
    addr.PostalCode = "700002"
    b.add(addr)
    pin.LossLocation = addr

    b.add(pin)
    out = pin
  })

  return out
}
